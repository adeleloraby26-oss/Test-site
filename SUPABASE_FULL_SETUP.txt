      WHERE c.id = conversation_id AND (c.user_a = auth.uid() OR c.user_b = auth.uid())

  );

    SELECT 1 FROM public.conversations c
    WHERE c.id = conversation_id AND (c.user_a = auth.uid() OR c.user_b = auth.uid())
  ));

-- ──────────────────────────────────────────────────────────
-- 18. FUNCTION: auto-update points → rank
-- ──────────────────────────────────────────────────────────
CREATE OR REPLACE FUNCTION public.update_user_rank()
RETURNS TRIGGER LANGUAGE plpgsql AS $$
BEGIN
  UPDATE public.users
  SET rank_id = (
    SELECT id FROM public.ranks
    WHERE min_points <= NEW.points AND max_points >= NEW.points
    ORDER BY min_points DESC LIMIT 1
  )
  WHERE id = NEW.id;
  RETURN NEW;
END;
$$;

DROP TRIGGER IF EXISTS trg_update_rank ON public.users;
CREATE TRIGGER trg_update_rank
  AFTER UPDATE OF points ON public.users
  FOR EACH ROW EXECUTE FUNCTION public.update_user_rank();

-- ──────────────────────────────────────────────────────────
-- 19. FUNCTION: create notification on rank up
-- ──────────────────────────────────────────────────────────
CREATE OR REPLACE FUNCTION public.notify_rank_up()
RETURNS TRIGGER LANGUAGE plpgsql AS $$
BEGIN
  IF NEW.rank_id IS DISTINCT FROM OLD.rank_id AND NEW.rank_id > OLD.rank_id THEN
    INSERT INTO public.notifications(user_id, type, title, body)
    SELECT NEW.id, 'rank_up',
           '🎉 Rank Up!',
           'You have reached a new rank: ' || (SELECT name FROM public.ranks WHERE id = NEW.rank_id);
  END IF;
  RETURN NEW;
END;
$$;

DROP TRIGGER IF EXISTS trg_notify_rank ON public.users;
CREATE TRIGGER trg_notify_rank
  AFTER UPDATE OF rank_id ON public.users
  FOR EACH ROW EXECUTE FUNCTION public.notify_rank_up();

-- ──────────────────────────────────────────────────────────
-- 20. FUNCTION: update conversation last_message
-- ──────────────────────────────────────────────────────────
CREATE OR REPLACE FUNCTION public.update_conversation_last_msg()
RETURNS TRIGGER LANGUAGE plpgsql AS $$
BEGIN
  UPDATE public.conversations
  SET last_message = NEW.content, last_msg_at = NEW.created_at
  WHERE id = NEW.conversation_id;
  RETURN NEW;
END;
$$;

DROP TRIGGER IF EXISTS trg_conversation_last_msg ON public.messages;
CREATE TRIGGER trg_conversation_last_msg
  AFTER INSERT ON public.messages
  FOR EACH ROW EXECUTE FUNCTION public.update_conversation_last_msg();
